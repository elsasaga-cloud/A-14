```python
def evaluate_trend(P, ma_dict):
    """
    P: 当前价格
    ma_dict: dict, 如 {"MA5":6.52, "MA10":6.48, ..., "MA90":6.22}
    """

    MA5 = ma_dict["MA5"]
    MA10 = ma_dict["MA10"]
    MA20 = ma_dict["MA20"]
    MA30 = ma_dict["MA30"]
    MA40 = ma_dict["MA40"]
    MA50 = ma_dict["MA50"]
    MA60 = ma_dict["MA60"]
    MA70 = ma_dict["MA70"]
    MA80 = ma_dict["MA80"]
    MA90 = ma_dict["MA90"]

    # ======================
    # 一、State 判断
    # ======================
    State = None

    if P < MA30 < MA60 < MA90:
        State = "0"

    elif P > MA30 > MA60 > MA90:
        State = "3"
        if P > MA10 > MA30 > MA50 > MA70 > MA90:
            State = "3+"

    elif P > MA30 and abs((P - MA60)/MA60) < 0.01:
        State = "2"
        if P > MA60:
            gaps = [
                abs((P - MA10)/MA10),
                abs((P - MA20)/MA20),
                abs((P - MA30)/MA30),
                abs((P - MA40)/MA40),
                abs((P - MA50)/MA50),
                abs((P - MA60)/MA60),
            ]
            max_gap = max(gaps)

            if max_gap <= 0.015:
                State = "2B-Compress"
            elif max_gap <= 0.03:
                State = "2B-EarlyBull"
            else:
                State = "2A"
        else:
            State = "2A"

    elif P > MA30 and P < MA60 and MA60 < MA90:
        State = "1"

    else:
        State = "X"

    # ======================
    # 二、CoreGap（位置）
    # ======================
    CoreGap = (P - MA60) / MA60

    if CoreGap <= 0.015:
        position_score = 100
    elif CoreGap <= 0.03:
        position_score = 80
    elif CoreGap <= 0.05:
        position_score = 60
    else:
        position_score = 30

    # ======================
    # 三、均线扩散度
    # ======================
    gaps = [
        abs((P - MA10)/MA10),
        abs((P - MA20)/MA20),
        abs((P - MA30)/MA30),
        abs((P - MA40)/MA40),
        abs((P - MA50)/MA50),
        abs((P - MA60)/MA60),
    ]
    avg_gap = sum(gaps) / len(gaps)

    if avg_gap <= 0.015:
        spread_score = 100
    elif avg_gap <= 0.03:
        spread_score = 80
    elif avg_gap <= 0.05:
        spread_score = 60
    else:
        spread_score = 40

    # ======================
    # 四、结构评分
    # ======================
    state_score_map = {
        "0": 0,
        "1": 20,
        "2A": 40,
        "2B-Compress": 70,
        "2B-EarlyBull": 80,
        "3": 90,
        "3+": 100,
        "X": 30
    }

    structure_score = state_score_map.get(State, 30)

    # ======================
    # 五、总评分
    # ======================
    total_score = (
        structure_score * 0.5 +
        position_score * 0.3 +
        spread_score * 0.2
    )

    # ======================
    # 六、交易建议
    # ======================
    if total_score >= 85:
        action = "主仓 / 顺势"
    elif total_score >= 70:
        action = "可参与"
    elif total_score >= 60:
        action = "观察 / 等回踩"
    else:
        action = "不参与"

    return {
        "State": State,
        "CoreGap": round(CoreGap, 4),
        "AvgGap": round(avg_gap, 4),
        "Score": round(total_score, 1),
        "Action": action
    }
```

```python
def calc_slope_score(S10, S20, S30):

    if S10 > 0 and S20 > 0 and S30 > 0:
```
