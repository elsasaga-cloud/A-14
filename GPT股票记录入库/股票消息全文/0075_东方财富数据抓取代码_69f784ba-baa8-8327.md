def get_stock_data(code="600398", klt=101, fqt=1):

    params = {
        "secid": f"1.{code}",

    result = []
    for k in klines:
        parts = k.split(",")

def panel_text_to_lines(panel_text):
    if not panel_text:
        return []

    arr = [x.strip() for x in panel_text.splitlines() if x.strip()]
    if not arr:
        return []

    pairs = []
    i = 0
    while i < len(arr):
        if i + 1 < len(arr):
            pairs.append(f"{arr[i]}: {arr[i + 1]}")
            i += 2
        else:
            pairs.append(arr[i])
            i += 1

    lines = []
    for j in range(0, len(pairs), 4):
        lines.append("\t".join(pairs[j:j + 4]))
    return lines

def extract_close_change_from_block(block):
    lines = [x.strip() for x in block.splitlines() if x.strip()]
    if not lines:
        return None, None

    close = None
    change_pct = None
    start_idx = 0

    for i, line in enumerate(lines):
        if line.startswith("股票数据"):
            start_idx = i + 1
            break

    skip_keywords = [
        "今开", "昨收", "最高价", "最低价", "换手率", "量比", "成交量", "成交额",
        "筹码分布", "5日涨幅", "MA", "Date:", "股票数据", "未找到",
    ]

    for line in lines[start_idx:]:
        if close is None:
            if not any(k in line for k in skip_keywords):
                v = parse_float(line)
                if v is not None:
                    close = v
                    continue
        else:
            if "%" in line:
                v = parse_float(line)
                if v is not None:
                    change_pct = v
                    break

    return close, change_pct

            "MA80": r"MA80[:：]?\s*([0-9.]+)",
            "MA90": r"MA90[:：]?\s*([0-9.]+)",
        }

        for k, p in pairs.items():
            m = re.search(p, block)
            if m:
                item[k] = parse_float(m.group(1))

        m = re.search(r"90%成本[:：]?\s*([0-9.]+)\s*-\s*([0-9.]+)", block)
        if m:
            item["cost_90_low"] = float(m.group(1))
            item["cost_90_high"] = float(m.group(2))

        m = re.search(r"70%成本[:：]?\s*([0-9.]+)\s*-\s*([0-9.]+)", block)
        if m:
            item["cost_70_low"] = float(m.group(1))
            item["cost_70_high"] = float(m.group(2))

        concs = list(re.finditer(r"集中度[:：]?\s*([0-9.]+)%", block))
        if len(concs) >= 1:
            item["conc_90"] = float(concs[0].group(1))
        if len(concs) >= 2:
            item["conc_70"] = float(concs[1].group(1))

        if item["date"] and item["close"] is not None:
            history.append(item)

    uniq = {}
    for row in history:
        uniq[row["date"]] = row

    result = list(uniq.values())
    result.sort(key=lambda x: x["date"])
    return result

def calc_easy_mode(P, MA60, MA90, prev_MA60, prev_MA90):
    gap60 = (MA60 - P) / P if P else None
    gap90 = (MA90 - P) / P if P else None
    slope60 = None if prev_MA60 in (None, 0) or MA60 is None else (MA60 - prev_MA60) / prev_MA60 * 100
    slope90 = None if prev_MA90 in (None, 0) or MA90 is None else (MA90 - prev_MA90) / prev_MA90 * 100

    if gap60 is None or gap90 is None:
        return "N/A", "长均线数据不足", None, None, slope60, slope90

    if ((gap60 > 0 and slope60 is not None and slope60 < 0) or
        (gap90 > 0 and slope90 is not None and slope90 < 0)):
        return "❌ 直接扔垃圾桶 (TRASH BIN)", "股价在长均线下且长均线拐头向下", gap60, gap90, slope60, slope90

    if (gap60 > 0.05) and (gap90 > 0.05):
        return "❌ 直接扔垃圾桶 (TRASH BIN)", "深水空头区", gap60, gap90, slope60, slope90
    elif (gap60 > 0) and (gap90 > 0):
        return "⚠ 谨慎进入下一步 (CAUTION)", "弱势震荡或突破前夜", gap60, gap90, slope60, slope90
    else:
        return "✅ 通过，进入详细评估 (PROCEED)", "趋势尚可", gap60, gap90, slope60, slope90


def calc_price_diff_method(P, row):
    if P is None or any(row.get(k) is None for k in ["MA5", "MA10", "MA20", "MA30"]):
        return {
            "D5": None, "D10": None, "D20": None, "D30": None,
            "score": 0,
            "state": "数据不足",
            "conclusion": "数据不足",
        }

    D5 = P - row["MA5"]
    D10 = P - row["MA10"]
    D20 = P - row["MA20"]
    D30 = P - row["MA30"]

    s5 = sign_diff(D5, row["MA5"])
    s10 = sign_diff(D10, row["MA10"])
    s20 = sign_diff(D20, row["MA20"])
    s30 = sign_diff(D30, row["MA30"])

    score = s5 + s10 + s20 + s30

    if [s5, s10, s20, s30] == [-1, -1, -1, -1]:
        state = "全面下破"
        conclusion = "空头不碰"
    elif [s5, s10, s20, s30] == [1, 1, 1, 1]:
        state = "全面上攻"
        conclusion = "多头才跟"
    elif [s5, s10, s20, s30] == [1, 1, -1, -1]:
        state = "短多长空"
        conclusion = "粘合才观"
    elif [s5, s10, s20, s30] == [1, -1, -1, -1]:
        state = "短暂反弹"
        conclusion = "空头未改"
    elif [s5, s10, s20, s30] == [-1, -1, 1, 1]:
        state = "下探支撑"
        conclusion = "粘合才观"
    elif [s5, s10, s20, s30] == [-1, 1, 1, 1]:
        state = "短回调但中期多头"
        conclusion = "多头健康调整"
    elif 0 in [s5, s10, s20, s30]:
        state = "临界状态"
        conclusion = "粘合观望"
    else:
        if score <= -3:
            state = "强空头"
            conclusion = "空头不碰"
        elif -2 <= score <= 0:
            state = "弱空头或粘合"
            conclusion = "粘合才观"
        elif 1 <= score <= 2:
            state = "多头初启"
            conclusion = "多头观察中"
        else:
            state = "稳定多头"
            conclusion = "多头才跟"

    return {
        "D5": D5, "D10": D10, "D20": D20, "D30": D30,
        "score": score,
        "state": state,
        "conclusion": conclusion,
    }


def calc_three_way(row):
    P = row["close"]
    mas = [row["MA5"], row["MA10"], row["MA20"], row["MA30"], row["MA40"], row["MA50"], row["MA60"], row["MA70"], row["MA80"], row["MA90"]]
    if any(v is None for v in mas):

    up_like = (
        P > row["MA60"] and P > row["MA90"] and

        row["MA30"] >= row["MA40"] >= row["MA50"] >= row["MA60"] and
        row["MA60"] >= row["MA70"] >= row["MA80"] >= row["MA90"]
    )

    down_like = (

        row["MA30"] <= row["MA40"] <= row["MA50"] <= row["MA60"] and

def calc_state_quant(row):
    P = row["close"]
    MA10 = row["MA10"]
    MA20 = row["MA20"]
    MA30 = row["MA30"]
    MA40 = row["MA40"]
    MA50 = row["MA50"]
    MA60 = row["MA60"]
    MA70 = row["MA70"]
    MA90 = row["MA90"]

    if None in [P, MA10, MA20, MA30, MA40, MA50, MA60, MA70, MA90]:
        return "X", None

    Gap10 = (P - MA10) / MA10
    Gap20 = (P - MA20) / MA20
    Gap30 = (P - MA30) / MA30
    Gap40 = (P - MA40) / MA40
    Gap50 = (P - MA50) / MA50
    Gap60 = (P - MA60) / MA60

    if P < MA30 < MA60 < MA90:
        return 0, None
    elif P > MA30 > MA60 > MA90:
        state = 3
        if P > MA10 > MA30 > MA50 > MA70 > MA90:
            state = "3+"
        return state, None
    elif P > MA30 and abs(Gap60) < 0.01:
        state = 2
        if P > MA30 and P > MA60:
            Gaps = [abs(Gap10), abs(Gap20), abs(Gap30), abs(Gap40), abs(Gap50), abs(Gap60)]
            MaxGap = max(Gaps)
            if MaxGap <= 0.015:
                return state, "2B-Compress"
            elif MaxGap <= 0.03:
                return state, "2B-EarlyBull"
            else:
                return state, "2A"
        else:
            return state, "2A"
    elif P > MA30 and P < MA60 and MA60 < MA90:
        return 1, None
    else:
        return "X", None


def calc_aei(history):
    result = []
    for i in range(len(history)):
        row = history[i]
        if i == 0:
            result.append({"date": row["date"], "AEI": 0.0})
            continue

        start = max(0, i - 20)
        prev_window = [x for x in history[start:i] if x.get("volume") is not None]
        meanV = None
        if prev_window:
            meanV = sum(x["volume"] for x in prev_window) / len(prev_window)

        conc_t = row.get("conc_90")
        conc_p = history[i - 1].get("conc_90")
        profit_t = row.get("profit_ratio")
        profit_p = history[i - 1].get("profit_ratio")
        vol_t = row.get("volume")

        deltaC = 0.0 if conc_t is None or conc_p is None else conc_t - conc_p
        deltaP = 0.0 if profit_t is None or profit_p is None else profit_t - profit_p

        if deltaC < 0.05:
            deltaC = 0.0
        if deltaP < 0.5:
            deltaP = 0.0

        Power = max(deltaC, deltaP * 0.4)

        if Power <= 0 or meanV in (None, 0) or vol_t is None:
            AEI = 0.0
        else:
            Effort = (vol_t / meanV) + 1e-6
            AEI = Power / Effort

        result.append({"date": row["date"], "AEI": AEI})

    alpha = 2 / (3 + 1)
    ema = None
    for i in range(len(result)):
        v = result[i]["AEI"]
        if ema is None:
            ema = v
        else:
            ema = alpha * v + (1 - alpha) * ema
        result[i]["AEI_EMA3"] = ema

    return result


def score_dimension(today, prev):
    chip = 3
    ma = 3
    vol = 3

    if today["close"] is not None and today["avg_cost"] is not None:
        if today["close"] > today["avg_cost"]:
            chip += 1
        else:
            chip -= 1

    if today.get("profit_ratio") is not None:
        if today["profit_ratio"] >= 20:
            chip += 1
        elif today["profit_ratio"] <= 5:
            chip -= 1

    if prev and today.get("conc_70") is not None and prev.get("conc_70") is not None:
        if today["conc_70"] < prev["conc_70"]:
            chip += 1
        elif today["conc_70"] > prev["conc_70"]:
            chip -= 1

    if today.get("MA5") is not None and today["close"] > today["MA5"]:
        ma += 1
    if today.get("MA10") is not None and today["close"] > today["MA10"]:
        ma += 1
    if today.get("MA20") is not None and today["close"] > today["MA20"]:
        ma += 1
    if today.get("MA60") is not None and today["close"] < today["MA60"]:
        ma -= 1
    if today.get("MA90") is not None and today["close"] < today["MA90"]:
        ma -= 1

    if today.get("volume_ratio") is not None:
        if 0.8 <= today["volume_ratio"] <= 1.5:
            vol += 1
        elif today["volume_ratio"] > 2.5:
            vol -= 1

    if today.get("turnover_rate") is not None:
        if 0.3 <= today["turnover_rate"] <= 2.0:
            vol += 1
        elif today["turnover_rate"] > 4:
            vol -= 1

    chip = max(1, min(5, chip))
    ma = max(1, min(5, ma))
    vol = max(1, min(5, vol))
    total = chip + ma + vol
    return chip, ma, vol, total


def trend_tag(today, prev):
    if prev is None:
        return "无趋势"

    P = today["close"]
    needed = ["MA5", "MA10", "MA20", "MA30", "MA40", "MA50", "MA60", "MA70", "MA80", "MA90"]
    if any(today.get(k) is None for k in needed):
        return "无趋势"

    if P > today["MA30"] > today["MA60"] > today["MA90"]:
        if today["MA5"] > today["MA10"] > today["MA20"] > today["MA30"]:
            return "趋势成型"
        return "趋势初启"

    spread = max(today[k] for k in needed) - min(today[k] for k in needed)
    if P and spread / P <= 0.015:
        return "无趋势"

    if P < today["MA30"] and P < today["MA60"] and P < today["MA90"]:
        return "趋势结束"

    return "趋势初启"

def ma_cluster_stats(row):
    keys = ["MA30", "MA40", "MA50", "MA60", "MA70", "MA80", "MA90"]
    vals = [row[k] for k in keys if row.get(k) is not None]
    if len(vals) < 3 or row["close"] in (None, 0):
        return None
    span = max(vals) - min(vals)
    span_pct = span / row["close"] * 100
    return {
        "span": span,
        "span_pct": span_pct,
        "tight": span_pct <= 0.6,
        "very_tight": span_pct <= 0.35,
        "vals": vals,
    }


def volume_context(history):
    if len(history) < 6:
        return {}
    today = history[-1]
    prev5 = history[-6:-1]
    vols = [x["volume"] for x in prev5 if x.get("volume") is not None]
    if not vols or today.get("volume") is None:
        return {}
    avg5 = sum(vols) / len(vols)
    ratio = today["volume"] / avg5 if avg5 else None
    return {
        "avg5": avg5,
        "ratio": ratio,
        "is_shrink": ratio is not None and ratio < 0.8,
        "is_deep_shrink": ratio is not None and ratio < 0.65,
        "is_expand": ratio is not None and ratio > 1.2,
    }


def chip_context(today, prev):
    ctx = {}
    if today.get("avg_cost") is not None and today.get("close") is not None and today["avg_cost"] != 0:
        ctx["under_avg"] = today["close"] < today["avg_cost"]
        ctx["above_avg"] = today["close"] > today["avg_cost"]
        ctx["premium"] = (today["close"] / today["avg_cost"] - 1) * 100

    if today.get("profit_ratio") is not None:
        ctx["profit_low"] = today["profit_ratio"] < 10
        ctx["profit_mid"] = 10 <= today["profit_ratio"] <= 40
        ctx["profit_high"] = today["profit_ratio"] > 40

    if prev and today.get("conc_70") is not None and prev.get("conc_70") is not None:
        ctx["conc70_change"] = today["conc_70"] - prev["conc_70"]
        ctx["conc_more_focus"] = today["conc_70"] < prev["conc_70"]
        ctx["conc_loosen"] = today["conc_70"] > prev["conc_70"]

    if prev and today.get("conc_90") is not None and prev.get("conc_90") is not None:
        ctx["conc90_change"] = today["conc_90"] - prev["conc_90"]

    if today.get("cost_70_low") is not None and today.get("cost_70_high") is not None and today.get("close") is not None:
        low = today["cost_70_low"]
        high = today["cost_70_high"]
        if high != low:
            pen = (today["close"] - low) / (high - low) * 100
            ctx["penetration"] = pen
            ctx["pen_weak"] = pen < 15
            ctx["pen_good"] = 20 <= pen <= 60
            ctx["pen_hot"] = pen > 80

    return ctx


def classify_regime(today, prev, history, state, sub_state, easy_action, diff_res, threeway, tag):
    mctx = ma_cluster_stats(today)
    vctx = volume_context(history)

    if easy_action.startswith("❌") and diff_res["score"] <= -2:
        return "deep_weak"

    if state in ["3+", 3]:
        if vctx.get("is_shrink") and diff_res["score"] >= 1:
            return "strong_trend_consolidation"
        return "strong_trend"

    if state == 2 and sub_state == "2B-Compress":
        return "compress_launch_zone"

    if state == 2 and sub_state == "2B-EarlyBull":
        return "early_bull_probe"

    if mctx and mctx["very_tight"] and vctx.get("is_deep_shrink"):
        return "shrink_wash_repair"

    if threeway == "粘合才观" and diff_res["state"] in ["短暂反弹", "临界状态", "弱空头或粘合"]:
        return "weak_repair"

    if state == 1:
        return "rebound_not_reverse"

    if tag == "无趋势":
        return "flat_wait"

    return "neutral_repair"


def get_penetration(today):
    if today.get("cost_70_low") is None or today.get("cost_70_high") is None:
        return None
    low = today["cost_70_low"]
    high = today["cost_70_high"]
    if high == low or today.get("close") is None:
        return None
    return (today["close"] - low) / (high - low) * 100


def get_premium(today):
    if today.get("avg_cost") in (None, 0) or today.get("close") is None:
        return None
    return (today["close"] / today["avg_cost"] - 1) * 100

def build_qa(history, stock_name, stock_code):
    history = filter_valid_history(history)
    if len(history) < 2:
        raise ValueError("有效历史不足2天，无法生成QA")

    today = history[-1]
    prev = history[-2] if len(history) >= 2 else None
    prev5 = history[-6] if len(history) >= 6 else None

    easy_action, easy_desc, gap60, gap90, slope60, slope90 = calc_easy_mode(
        today["close"], today["MA60"], today["MA90"],
        prev["MA60"] if prev else None,
        prev["MA90"] if prev else None,
    )

    diff_res = calc_price_diff_method(today["close"], today)
    threeway = calc_three_way(today)
    state, sub_state = calc_state_quant(today)
    chip_score, ma_score, vol_score, total_score = score_dimension(today, prev)
    tag = trend_tag(today, prev)

    aei = calc_aei(history)
    aei20 = aei[-20:] if len(aei) >= 20 else aei[:]
    aei5sum = sum(x["AEI"] for x in aei[-5:]) if len(aei) >= 5 else sum(x["AEI"] for x in aei)
    aei_today = aei[-1] if aei else None

    penetration = get_penetration(today)
    premium = get_premium(today)

    trend = None
    if today["avg_cost"] is not None and today["avg_cost"] != 0:
        trend = today["close"] > today["avg_cost"]

    signal = "HOLD"
    desc = "震荡/观察"
    if trend and penetration is not None and (20 <= penetration <= 80) and today["volume_ratio"] is not None and today["volume_ratio"] >= 1.0:
        signal = "BUY"
        desc = "趋势启动，放量突破"
    elif (trend is False) and ((penetration is not None and penetration < 10) or (premium is not None and premium < -3.0)):
        signal = "SELL"
        desc = "趋势破位，筹码溃散"
    else:
        signal = "HOLD"
        if trend and penetration is not None and penetration > 80:
            desc = "高位锁仓 (只卖不买)"
        elif trend is False:
            desc = "水下蓄势 (未破位)"
        else:
            desc = "中继震荡"

    if aei_today:
        if aei_today["AEI"] > 3.0:
            aei_desc = "极强"
        elif aei_today["AEI"] >= 1.5:
            aei_desc = "强"
        elif aei_today["AEI"] >= 0.5:
            aei_desc = "中等"
        elif aei_today["AEI"] == 0:
            aei_desc = "中性/锁仓"
        else:
            aei_desc = "弱"
    else:
        aei_desc = "数据不足"

    if aei5sum > 4.5:
        aei_trend = "真吸筹趋势"
    elif 2.5 <= aei5sum <= 4.5:
        aei_trend = "弱真"
    else:
        aei_trend = "假吸筹/中性"

    ma5_slope = None
    ma10_slope = None
    if prev and prev.get("MA5") not in (None, 0) and today.get("MA5") is not None:
        ma5_slope = (today["MA5"] - prev["MA5"]) / prev["MA5"] * 100
    if prev and prev.get("MA10") not in (None, 0) and today.get("MA10") is not None:
        ma10_slope = (today["MA10"] - prev["MA10"]) / prev["MA10"] * 100

    slopes = {}
    for k in ["MA5", "MA10", "MA20", "MA30", "MA40", "MA50", "MA60", "MA70", "MA80", "MA90"]:
        slopes[k] = None
        if prev and prev.get(k) not in (None, 0) and today.get(k) is not None:
            slopes[k] = (today[k] - prev[k]) / prev[k] * 100

    dev5 = ((today["close"] - today["MA5"]) / today["MA5"] * 100) if today.get("MA5") not in (None, 0) else None

    support_low, support_high, pressure_low, pressure_high = get_support_pressure(today)
    balance_low = min(today["MA30"], today["avg_cost"]) if today["avg_cost"] is not None and today["MA30"] is not None else today["MA30"]
    balance_high = max(today["close"], today["avg_cost"]) if today["avg_cost"] is not None else today["close"]

    q1 = q1_dynamic(today, prev, history, state, sub_state, easy_action, diff_res, threeway, tag)
    q2 = q2_dynamic(today, prev, history, state, sub_state, easy_action, diff_res, threeway, tag)
    q3 = q3_dynamic(today, prev, history, state, sub_state, easy_action, diff_res, threeway, tag)
    q4 = q4_dynamic(today, prev, history, state, sub_state, threeway, aei5sum)
    q5 = q5_dynamic(today, prev, history, aei_trend, aei_today, aei5sum, penetration, premium, trend)
    q6 = q6_dynamic(today, prev, history, state, sub_state, penetration, premium, trend, aei_today, aei5sum)

    lines = []
    lines.append(f"股票：{stock_name} {stock_code}")
    lines.append(f"日期：{today['date']}")
    lines.append("")
    lines.append("1. 目前走势是否健康？")
    lines.append(q1)
    lines.append("")
    lines.append("2. State含义及操作建议")
    lines.append(q2)
    lines.append("")
    lines.append("3. 后续走势预判")
    lines.append(q3)
    lines.append("")
    lines.append("4. 还要震荡多久？")
    lines.append(q4)
    lines.append("")
    lines.append("5. 主力吸筹趋势（基于AEI）")
    lines.append(q5)
    lines.append("")
    lines.append("6. 是否准备拉升？")
    lines.append(q6)
    lines.append("")
    lines.append("风险提示")
    lines.append("1. 若股价重新跌破中期关键均线且量能同步放大，则修复结构可能失效。")
    lines.append("2. 若获利比例持续走弱且集中度恶化，说明筹码稳定性仍不足。")
    lines.append("3. 若出现放量但价格无法上行，需警惕高位换手而非有效突破。")
