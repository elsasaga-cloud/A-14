def load_and_clean_data(file_path: str) -> pd.DataFrame:
    try:
        df = pd.read_excel(file_path, skiprows=3)
    except FileNotFoundError:
        raise FileNotFoundError(f"找不到文件: {file_path}")

    df.columns = df.columns.str.strip()
    df = df.dropna(how='all').reset_index(drop=True)

    required_cols = ['日期', '开盘', '收盘', '最高', '最低', '成交量']
    missing = [c for c in required_cols if c not in df.columns]
    if missing:
        raise ValueError(f"Excel缺少必需列: {missing}\n实际列名: {df.columns.tolist()}")

    df['日期'] = pd.to_datetime(df['日期'], errors='coerce')
    df = df.dropna(subset=['日期'])
    df = df.sort_values('日期').reset_index(drop=True)

    # 所有非日期列强制转数值
    for col in df.columns:
        if col == '日期':
            continue
        df[col] = pd.to_numeric(df[col], errors='coerce')

    # 衍生列
    df['日收益率']   = df['收盘'].pct_change() * 100
    df['日涨跌额']   = df['收盘'].diff()
    df['真实振幅']   = (df['最高'] - df['最低']) / df['收盘'].shift(1) * 100
    df['log收益率']  = np.log(df['收盘'] / df['收盘'].shift(1)) * 100
    df['相对成交量'] = df['成交量'] / df['成交量'].rolling(20).mean()

    print(f"  ✅ 数据读取成功")
    print(f"  日期范围: {df['日期'].min().strftime('%Y-%m-%d')} ~ "
          f"{df['日期'].max().strftime('%Y-%m-%d')}")
    print(f"  总行数  : {len(df)}")
    print(f"  列数    : {len(df.columns)}")
    return df

def build_event_set(df, vol_thresh, ret_thresh):
    tmp = df.copy()
    tmp['_vol_ma20']    = tmp['成交量'].rolling(20).mean()
    tmp['_vol_ratio20'] = tmp['成交量'] / tmp['_vol_ma20']
    mask = (tmp['_vol_ratio20'] > vol_thresh) & (tmp['日收益率'].abs() > ret_thresh)
    return set(tmp.loc[mask, '日期'])

)

def classify_day(date, known_earnings_dt, auto_event_set, tol=2):
    for ed in known_earnings_dt:
        if abs((date - ed).days) <= tol:
            return '财报日'
    if date in auto_event_set:
        return '突发事件日'
    return '常规日'

def fetch_option_data(ticker_str, current_price):
    result = {
        'success'       : False,
        'current_iv'    : None,
        'options_dates' : [],
        'call_chain'    : None,
        'ex_div_date'   : None,
        'dividend_yield': None,
    }
    try:
        tk   = yf.Ticker(ticker_str)
        info = tk.info
        result['dividend_yield'] = info.get('dividendYield', None)

        ex_raw = info.get('exDividendDate', None)
        if ex_raw is not None:
            try:
                if isinstance(ex_raw, (int, float)):
                    result['ex_div_date'] = datetime.utcfromtimestamp(ex_raw).strftime('%Y-%m-%d')
                else:
                    result['ex_div_date'] = pd.to_datetime(ex_raw).strftime('%Y-%m-%d')
            except Exception:
                result['ex_div_date'] = None

        opt_dates = tk.options
        result['options_dates'] = list(opt_dates)

        if opt_dates:
            iv_list = []
            for exp_date in opt_dates[:3]:
                chain = tk.option_chain(exp_date)
                calls = chain.calls.dropna(subset=['impliedVolatility']).copy()
                atm   = calls[
                    (calls['strike'] >= current_price * 0.95) &
                    (calls['strike'] <= current_price * 1.05)
                ]
                if not atm.empty:
                    iv_list.extend(atm['impliedVolatility'].tolist())
                    if result['call_chain'] is None:
                        result['call_chain']           = calls.copy()
                        result['call_chain']['expiry'] = exp_date

            if iv_list:
                result['current_iv'] = float(np.mean(iv_list)) * 100
                result['success']    = True

        print(f"  ✅ IV获取成功: {result['current_iv']:.1f}%")
        if result['ex_div_date']:
            print(f"  除息日  : {result['ex_div_date']}")
        if result['dividend_yield']:
            print(f"  股息率  : {result['dividend_yield']*100:.2f}%")

    except Exception as e:
        print(f"  ⚠️  IV获取失败 ({e}), 将使用历史波动率替代")

    return result

def compute_prob_matrix(df, dte_list, otm_list, use_high=False, exclude_earnings=True):
    closes = df['收盘'].values
    highs  = df['最高'].values
    is_er  = (df['交易日类型'] == '财报日').values
    n      = len(df)
    mat    = np.full((len(dte_list), len(otm_list)), np.nan)

    for i, dte in enumerate(dte_list):
        if n <= dte:
            continue
        samples = []
        for k in range(n - dte):
            if exclude_earnings and is_er[k:k+dte].any():
                continue
            sp = closes[k]
            if sp <= 0 or np.isnan(sp):
                continue
            val = (highs[k:k+dte].max()/sp - 1)*100 if use_high else (closes[k+dte]/sp - 1)*100
            samples.append(val)

        if samples:
            arr = np.array(samples)
            for j, otm in enumerate(otm_list):
                mat[i, j] = (arr > otm).mean() * 100
    return mat

def backtest_cc(df, otm_pct, dte=30, exclude_earnings=True, normal_hv=0.30):
    closes = df['收盘'].values
    is_er  = (df['交易日类型'] == '财报日').values
    hv20   = df['HV20'].values
    dates  = df['日期'].values
    n      = len(df)
    trades = []
    i      = 0

    while i <= n - dte - 1:
        window_er = is_er[i:i+dte]
        if exclude_earnings and window_er.any():
            first_er_offset = int(np.argmax(window_er))
            i += first_er_offset + 1
            continue

        sp     = float(closes[i])
        strike = sp * (1 + otm_pct / 100)
        ep     = float(closes[i + dte])

        hv_now   = float(hv20[i]) if not np.isnan(hv20[i]) else normal_hv
        iv_est   = hv_now * 1.1
        otm_disc = max(0.3, 1 - abs(otm_pct) * 0.08)
        premium  = sp * iv_est * np.sqrt(dte / 252) * otm_disc * 0.5

        called = ep >= strike
        profit = (strike - sp + premium) if called else premium

        trades.append({
            '开仓日'  : pd.Timestamp(dates[i]).strftime('%Y-%m-%d'),
            '到期日'  : pd.Timestamp(dates[i + dte]).strftime('%Y-%m-%d'),
            '开仓价'  : round(sp, 4),
            'Strike'  : round(strike, 4),
            '到期价'  : round(ep, 4),
            '权利金'  : round(premium, 4),
            '盈亏/股' : round(profit, 4),
            '收益率%' : round(profit / sp * 100, 4),
            '被行权'  : called,
            'OTM%'    : otm_pct,
        })
        i += dte

    return pd.DataFrame(trades)

def print_bt_row(label, bt):
    if len(bt) == 0:
        print(f"  {label:<8}  无有效交易")
        return
    wins   = (bt['盈亏/股'] > 0).mean() * 100
    called = bt['被行权'].mean() * 100
    avg    = bt['收益率%'].mean()
    total  = bt['收益率%'].sum()
    mxls   = bt['收益率%'].min()
    std_   = bt['收益率%'].std()
    sharpe = (avg / std_ * np.sqrt(12)) if std_ > 0 else 0
    print(f"  {label:<8} {len(bt):>5}次  "
          f"胜率{wins:>5.1f}%  被行权{called:>5.1f}%  "
          f"均收益{avg:>+6.3f}%  总收益{total:>+8.1f}%  "
          f"最大亏{mxls:>+6.3f}%  Sharpe{sharpe:>5.2f}")

def format_worksheet(ws, is_backtest: bool = False):
    """
    对单个worksheet应用统一格式:
    - 字体: Times New Roman, 10号
    - 对齐: 左对齐, 垂直居中
    - 列宽: 自动适配 (最小8, 最大50)
    - 回测sheet的 开仓日/到期日 列宽固定为20
    """
    BASE_FONT      = Font(name='Times New Roman', size=10)
    BASE_ALIGNMENT = Alignment(horizontal='left', vertical='center', wrap_text=False)

    # ── 找出回测sheet中日期列的列号 ────────────────────────────
    date_col_indices = set()
    if is_backtest:
        for cell in ws[1]:                          # 第1行是表头
            if cell.value in DATE_COLS_IN_BT:
                date_col_indices.add(cell.column)  # 1-based

    # ── 逐行应用字体和对齐 ─────────────────────────────────────
    for row in ws.iter_rows():
        for cell in row:
            cell.font      = BASE_FONT
            cell.alignment = BASE_ALIGNMENT

    # ── 列宽计算 ───────────────────────────────────────────────
    for col_cells in ws.columns:
        col_letter = get_column_letter(col_cells[0].column)
        col_idx    = col_cells[0].column            # 1-based

        # 回测sheet的日期列: 固定宽度20
        if is_backtest and col_idx in date_col_indices:
            ws.column_dimensions[col_letter].width = 20
            continue

        # 其他列: 按内容自动适配
        max_len = 0
        for cell in col_cells:
            if cell.value is not None:
                # 中文字符算2个宽度单位
                val_str  = str(cell.value)
                char_len = sum(2 if ord(c) > 127 else 1 for c in val_str)
                max_len  = max(max_len, char_len)

        # 留1个字符padding, 限制在[8, 50]
        fitted_width = min(max(max_len + 1, 8), 50)
        ws.column_dimensions[col_letter].width = fitted_width

    # ── 冻结首行 (方便阅读) ────────────────────────────────────
    ws.freeze_panes = 'A2'

# ── 10-C 打开workbook应用格式 ─────────────────────────────────
