## 🔧 第一步：先试「系统自带下载器」（完全绕开浏览器）
浏览器点下载没反应，咱们直接用系统自带的工具下载，它不受浏览器扩展或安全策略影响：

1.  按 `Win + R`，输入 `cmd` 回车，打开命令提示符
2.  输入下面这条命令，下载一个测试文件（比如 1MB 的测速文件）：
    ```cmd
    bitsadmin /transfer myDownloadJob /download /priority normal https://speed.hetzner.de/1MB.bin C:\Users\simon\Downloads\test.bin
    ```
3.  回车后，如果命令行开始显示进度条，说明下载功能是好的，只是浏览器出了问题；如果直接报错，说明是系统级别的问题。

---

## 🛡️ 第二步：排查「系统策略/安全软件拦截」
如果上面的命令也报错，说明系统里有东西在拦所有下载行为：

### 1. 先临时退出所有第三方安全软件
- 360、火绒、腾讯电脑管家、鲁大师等，全部右键退出
- 再试一次上面的 `bitsadmin` 命令，如果能下载了，就是安全软件的「下载保护」功能在搞鬼，进去把它关掉就行。

### 2. 检查组策略是否被篡改
1.  按 `Win + R`，输入 `gpedit.msc` 回车，打开组策略编辑器
2.  导航到：`用户配置 → 管理模板 → Windows 组件 → Internet Explorer → 安全功能 → 限制文件下载`
3.  看看右边的「限制文件下载」是不是被设为「已启用」，如果是，改成「未配置」或「已禁用」
4.  重启电脑再试下载

---

## 🧹 第三步：重置浏览器和网络设置
如果 `bitsadmin` 能下载，只是浏览器不行，咱们把浏览器彻底重置：

### 1. 重置 Chrome/Edge 浏览器
- 打开浏览器设置 → 找到「重置设置」
- 选择「将设置恢复为原始默认值」，确认重置
- 重置完成后，关闭所有浏览器窗口，重新打开再试下载

### 2. 重置网络组件（防止代理/网络设置锁死下载）
以管理员身份打开 PowerShell，运行：
```powershell
netsh winsock reset
netsh int ip reset
ipconfig /flushdns
```
运行完重启电脑，再试浏览器下载。

def panel_text_to_lines(panel_text):

    arr = [x.strip() for x in panel_text.splitlines() if x.strip()]
    if not arr:
        return []

    pairs = []
    i = 0
    while i < len(arr):
        if i + 1 < len(arr):
            pairs.append(f"{arr[i]}: {arr[i + 1]}")
            i += 2
        else:
            pairs.append(arr[i])
            i += 1

    lines = []
    for j in range(0, len(pairs), 4):
        lines.append("\t".join(pairs[j:j + 4]))
    return lines

    if today.get("cost_70_low") is not None and today.get("cost_70_high") is not None and today.get("close") is not None:
        low = today["cost_70_low"]
        high = today["cost_70_high"]
        if high != low:
            pen = (today["close"] - low) / (high - low) * 100
            ctx["penetration"] = pen
            ctx["pen_weak"] = pen < 15
            ctx["pen_good"] = 20 <= pen <= 60
            ctx["pen_hot"] = pen > 80

    return ctx


def classify_regime(today, prev, history, state, sub_state, easy_action, diff_res, threeway, tag):
    mctx = ma_cluster_stats(today)
    vctx = volume_context(history)

    if easy_action.startswith("❌") and diff_res["score"] <= -2:
        return "deep_weak"

    if state in ["3+", 3]:
        if vctx.get("is_shrink") and diff_res["score"] >= 1:
            return "strong_trend_consolidation"
        return "strong_trend"

    if state == 2 and sub_state == "2B-Compress":
        return "compress_launch_zone"

    if state == 2 and sub_state == "2B-EarlyBull":
        return "early_bull_probe"

    if mctx and mctx["very_tight"] and vctx.get("is_deep_shrink"):
        return "shrink_wash_repair"

    if threeway == "粘合才观" and diff_res["state"] in ["短暂反弹", "临界状态", "弱空头或粘合"]:
        return "weak_repair"

    if state == 1:
        return "rebound_not_reverse"

    if tag == "无趋势":
        return "flat_wait"

    return "neutral_repair"


def get_penetration(today):
    if today.get("cost_70_low") is None or today.get("cost_70_high") is None:
        return None
    low = today["cost_70_low"]
    high = today["cost_70_high"]
    if high == low or today.get("close") is None:
        return None
    return (today["close"] - low) / (high - low) * 100


def get_premium(today):
    if today.get("avg_cost") in (None, 0) or today.get("close") is None:
        return None
    return (today["close"] / today["avg_cost"] - 1) * 100


def get_support_pressure(today):
    support_low = today["cost_70_low"] if today.get("cost_70_low") is not None else today.get("MA60")
    support_high = today.get("MA60")

    arr = [today.get("avg_cost"), today.get("MA20"), today.get("MA30")]
    valid = [x for x in arr if x is not None]
    pressure_low = max(valid) if valid else None

    pressure_high = today["cost_70_high"] if today.get("cost_70_high") is not None else today.get("MA10")
    return support_low, support_high, pressure_low, pressure_high


def get_volume_ratio_5(history):
    if len(history) < 6:
        return None
    today = history[-1]
    prev5 = history[-6:-1]
    vols = [x["volume"] for x in prev5 if x.get("volume") is not None]
    if not vols or today.get("volume") is None:
        return None
    avg5 = sum(vols) / len(vols)
    if avg5 == 0:
        return None
    return today["volume"] / avg5


def get_ma_cross_bias(today):
    if today.get("MA5") is None or today.get("MA10") is None or today["MA10"] == 0:
        return None
    diff = today["MA5"] - today["MA10"]
    if abs(diff) / today["MA10"] <= 0.002:
        return "MA5与MA10处于粘合状态"
    return "MA5位于MA10上方" if diff > 0 else "MA5位于MA10下方"


def make_trigger_sentence(today):
    support_low, support_high, pressure_low, pressure_high = get_support_pressure(today)
    arr = []

    if pressure_high is not None:
        arr.append(f"向上若放量突破 {format_num(pressure_high)}")
    elif pressure_low is not None:
        arr.append(f"向上若有效突破 {format_num(pressure_low)}")

    if support_low is not None:
        arr.append(f"向下若有效跌破 {format_num(support_low)}")
    elif support_high is not None:
        arr.append(f"向下若失守 {format_num(support_high)}")

    return "；".join(arr) if arr else "等待关键价位突破后确认方向"


def aei_event_summary(history, topn=2):
    aei = calc_aei(history)
    rows = []
    for i, x in enumerate(aei):
        if i == 0:
            continue
        rows.append((x["date"], x["AEI"]))
    rows.sort(key=lambda t: t[1], reverse=True)
    rows = [x for x in rows if x[1] > 0][:topn]
    if not rows:
        return "近阶段未观察到特别突出的AEI爆发点"
    return "、".join([f"{d}(AEI={format_num(v, 2)})" for d, v in rows])

def build_qa(history, stock_name, stock_code):
    history = filter_valid_history(history)
    if len(history) < 2:
        raise ValueError("有效历史不足2天，无法生成QA")

    today = history[-1]
    prev = history[-2] if len(history) >= 2 else None
    prev5 = history[-6] if len(history) >= 6 else None

    easy_action, easy_desc, gap60, gap90, slope60, slope90 = calc_easy_mode(
        today["close"], today["MA60"], today["MA90"],
        prev["MA60"] if prev else None,
        prev["MA90"] if prev else None,
    )

    diff_res = calc_price_diff_method(today["close"], today)
    threeway = calc_three_way(today)
    state, sub_state = calc_state_quant(today)
    chip_score, ma_score, vol_score, total_score = score_dimension(today, prev)
    tag = trend_tag(today, prev)

    aei = calc_aei(history)
    aei20 = aei[-20:] if len(aei) >= 20 else aei[:]
    aei5sum = sum(x["AEI"] for x in aei[-5:]) if len(aei) >= 5 else sum(x["AEI"] for x in aei)
    aei_today = aei[-1] if aei else None

    penetration = get_penetration(today)
    premium = get_premium(today)

    trend = None
    if today["avg_cost"] is not None and today["avg_cost"] != 0:
        trend = today["close"] > today["avg_cost"]

    signal = "HOLD"
    desc = "震荡/观察"
    if trend and penetration is not None and (20 <= penetration <= 80) and today["volume_ratio"] is not None and today["volume_ratio"] >= 1.0:
        signal = "BUY"
        desc = "趋势启动，放量突破"
    elif (trend is False) and ((penetration is not None and penetration < 10) or (premium is not None and premium < -3.0)):
        signal = "SELL"
        desc = "趋势破位，筹码溃散"
    else:
        signal = "HOLD"
        if trend and penetration is not None and penetration > 80:
            desc = "高位锁仓 (只卖不买)"
        elif trend is False:
            desc = "水下蓄势 (未破位)"
        else:
            desc = "中继震荡"

    if aei_today:
        if aei_today["AEI"] > 3.0:
            aei_desc = "极强"
        elif aei_today["AEI"] >= 1.5:
            aei_desc = "强"
        elif aei_today["AEI"] >= 0.5:
            aei_desc = "中等"
        elif aei_today["AEI"] == 0:
            aei_desc = "中性/锁仓"
        else:
            aei_desc = "弱"
    else:
